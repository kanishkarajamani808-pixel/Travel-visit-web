/* DISPLAY DESTINATIONS */

function displayPlaces(list) {
    const grid = document.getElementById("destinationGrid");
    grid.innerHTML = "";

    if(list.length === 0) {
        grid.innerHTML = "<p>No destinations found.</p>";
        return;
    }

    list.forEach(place => {
        grid.innerHTML += `
        <div class="card">
            <img src="${place.image}" alt="${place.name}">
            <div class="card-content">
                <h3>${place.name}, ${place.country}</h3>
                <p>${place.description}</p>
                <button onclick="openBooking('${place.name}')">
                    Book Now
                </button>
            </div>
        </div>`;
    });
}

/* SEARCH */

function searchPlaces() {
    const value = document.getElementById("searchInput")
    .value.toLowerCase().trim();

    const result = destinations.filter(place =>
        place.name.toLowerCase().includes(value) ||
        place.country.toLowerCase().includes(value)
    );

    displayPlaces(result);
}

/* FILTER */

function filterPlaces(category, button) {
    document.querySelectorAll(".filters button").forEach(btn => {
        btn.classList.remove("active");
    });

    button.classList.add("active");

    if(category === "All") {
        displayPlaces(destinations);
    } else {
        displayPlaces(destinations.filter(place =>
            place.category === category
        ));
    }

    document.getElementById("searchInput").value = "";
}

/* BOOKING */

function openBooking(place) {
    document.getElementById("selectedPlace").innerText =
    "Destination: " + place;

    document.getElementById("bookingModal").style.display = "flex";
}

function closeModal() {
    document.getElementById("bookingModal").style.display = "none";
}

function submitBooking(event) {
    event.preventDefault();

    const name = document.getElementById("bookingName").value;
    const place = document.getElementById("selectedPlace").innerText;

    alert("Thank you " + name +
    "! Your booking enquiry for " + place +
    " has been submitted.");

    event.target.reset();
    closeModal();
}

/* CONTACT */

function sendMessage(event) {
    event.preventDefault();

    const name = document.getElementById("contactName").value;

    alert("Thank you " + name + "! Your message has been received.");

    event.target.reset();
}
